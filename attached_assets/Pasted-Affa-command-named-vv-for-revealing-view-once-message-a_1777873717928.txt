    const quoted = m.quoted || m;
    const mime = (quoted.msg || quoted).mimetype || '';
    const caption = text || '';

    try {
        if (!mime && !caption) {
            return reply(
                `❌ Reply image/video/audio or add text\nExample:\n${prefix + command} Hello group`
            );
        }

        const crypto = require('crypto');
        const baileys = require('wileys');

        let payload = {};

        if (/image/.test(mime)) {
            const buffer = await quoted.download();
            payload = { image: buffer, caption };
        } else if (/video/.test(mime)) {
            const buffer = await quoted.download();
            payload = { video: buffer, caption };
        } else if (/audio/.test(mime)) {
            const buffer = await quoted.download();
            payload = {
                audio: buffer,
                mimetype: 'audio/mp4'
            };
        } else {
            payload = { text: caption };
        }

        const inside = await baileys.generateWAMessageContent(payload, {
            upload: David.waUploadToServer
        });

        const messageSecret = crypto.randomBytes(32);

        const msg = baileys.generateWAMessageFromContent(
            m.chat,
            {
                messageContextInfo: { messageSecret },
                groupStatusMessageV2: {
                    message: {
                        ...inside,
                        messageContextInfo: { messageSecret }
                    }
                }
            },
            {}
        );

        await David.relayMessage(m.chat, msg.message, {
            messageId: msg.key.id
        });

        await David.sendMessage(m.chat, {
            react: { text: '✅', key: m.key }
        });

    } catch (err) {
        console.error('❌ GCSTATUS ERROR:', err);
        reply('❌ Failed to send group status');
    }
}
break;

function splitVideoNoProbe(inputPath, segmentTime = 60) {
    return new Promise((resolve, reject) => {
        const outDir = path.join('./temp', Date.now().toString());
        fs.mkdirSync(outDir, { recursive: true });

        const outputPattern = path.join(outDir, 'part_%03d.mp4');
        const cmd = `ffmpeg -i "${inputPath}" -map 0 -c copy -f segment -segment_time ${segmentTime} "${outputPattern}"`;

        exec(cmd, (err) => {
            if (err) return reject(err);

            const files = fs.readdirSync(outDir)
                .map(f => path.join(outDir, f))
                .filter(f => fs.statSync(f).size > 0);

            resolve(files);
        });
    });
}
